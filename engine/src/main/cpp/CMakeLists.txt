cmake_minimum_required(VERSION 3.22.1)
project(spektralab_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_NATIVE_RAW OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_HOST OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_DESKTOP OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_BUILD_TEST_DLL OFF CACHE BOOL "" FORCE)
set(SPEKTRALAB_REQUIRE_VULKAN ON CACHE BOOL "" FORCE)
set(SPEKTRALAB_VULKAN_HEADERS "${CMAKE_SYSROOT}/usr/include" CACHE PATH "" FORCE)
set(UPSTREAM "${CMAKE_CURRENT_LIST_DIR}/../../../../third_party/spektralab")
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
# Keep local account and workspace paths out of assertions and debug metadata.
add_compile_options("-ffile-prefix-map=${PORT_ROOT}=."
                    "-ffile-prefix-map=${ANDROID_NDK}=android-ndk")
add_subdirectory("${UPSTREAM}/engine" upstream-engine)
add_library(spektralab_android SHARED native_bridge.cpp)
target_include_directories(spektralab_android PRIVATE "${UPSTREAM}/engine/include")
target_link_libraries(spektralab_android PRIVATE spektralab_engine jnigraphics android log)
# Preserve upstream's no-contraction maths and Android's 16 KB page compatibility.
target_compile_options(spektralab_android PRIVATE -ffp-contract=off -fno-fast-math -Wall -Wextra)
target_link_options(spektralab_android PRIVATE "-Wl,-z,max-page-size=16384")
